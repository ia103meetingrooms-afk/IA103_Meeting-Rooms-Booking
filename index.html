function renderHomeView(main) { // 1. กรองรายการจองทั้งหมดที่ตรงกับวันที่เลือกไว้ก่อน const dayBookings = state.bookings.filter(b => b.date === state.selectedDate); const scaleHours = []; for (let h = state.startHour; h <= state.endHour; h++) { scaleHours.push(String(h).padStart(2, '0') + ':00'); } const timeScaleHtml = `
${scaleHours.map(t => `${t}`).join('')}
`; // 2. ดึงรายการจองของแต่ละห้องเฉพาะห้องนั้นๆ มาวาด Timeline const roomRows = state.rooms.map(room => { const rBookings = dayBookings.filter(b => String(b.roomId) === String(room.id)); return `
${escapeHtml(room.name)} (${room.capacity} ${t('cap')})
${timeScaleHtml}
${renderTimelineSegments(rBookings)}
`; }).join(''); main.innerHTML = `

${t('homeTimelineTitle')}

${roomRows || `

${t('noRooms')}

`}
`; }
